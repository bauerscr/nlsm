\section{Block spin transforms}
\label{sec:block-gauss}

This explains roughly Section~3.2.
%
The lattice $\Lambda$ has $n_{\rm fine}=|\Lambda|$ sites, $E$ oriented edges, and $B$ blocks.
Fields on the lattice are denoted $\phi$ and block spin fields $V$.

\subsection{Nonlinear block spin transforms}

The preprints use `soft' block spin transforms defined in terms of a penalization parameter $a>0$.
For a block $Y \subset \Lambda$, the linear block average of $\phi: \Lambda \to \R$ is defined by
\begin{equation}
 (Q\phi)_Y=\sum_{x\in Y}w_Y(x)\phi_x,
 \qquad w_Y(x)\ge0,\qquad \sum_{x\in Y}w_Y(x)=1.
\end{equation}
The nonlinear block spin average $\widehat m_Y(q)$ of a spin configuration $q: \Lambda \to S^2$ is then defined by
\begin{equation} \label{eq:hatm}
  m_Y(q) = (Qq)_Y, \qquad \widehat m_Y(q) =
  \begin{cases}
      \frac{m_Y(q)}{|m_Y(q)|} & (|m_Y(q)|\geq \frac12)
      \\
      \dots
  \end{cases}
\end{equation}
The block spin transform of a measure on configurations $q: \Lambda \to S^2$ is done by inserting the following factor into the measure for $q$:
\begin{equation}
\propto \prod_{Y \in \Lambda'} e^{-\frac{a}{2}|V_Y-\widehat m_Y(q)|^2} \, d\omega(V_Y)
\end{equation}
and integrating over $q$. This defines a new measure on block configurations $V :\Lambda' \to S^2$ where $Y\in\Lambda'$ are the blocks of the original lattice $\Lambda$.


\subsection{Gaussian block spin transforms}

For Gaussian measures, the block spin transform is an explicit recursion for the Gaussian functional (called `precision' in the preprints),
when the linear block average $Q\phi$ is used instead of $\widehat m$.

The kinetic term (or precision matrix) of the initial Gaussian measure is given by the Laplacian on the lattice $\Lambda$ [O3, above (3.4)]:
\begin{equation}
  K_{\varnothing} = d^* d.
\end{equation}
For a precision parameter $a>0$, the next-scale kinetic term is defined by [O3, (3.4)]
\begin{equation}
K_{h+1} = a\Id-a^2Q(K_h+aQ^*Q)^{-1}Q^*.
\end{equation}
This choice is explained by the following well-known fact.

\begin{proposition}
For a given block spin field $V\in\R^B$, 
\begin{equation} \label{eq:Gauss-var}
 \frac12\langle V,K_{h+1}V\rangle
 =\min_{\phi\in\R^{n_{\rm fine}}} \mathcal E_V(\phi)
 =\min_{\phi\in\R^{n_{\rm fine}}}
 \left\{\frac12\langle\phi,K_h\phi\rangle
              +\frac a2\|V-Q\phi\|^2\right\}.
\end{equation}
Here $n_{\rm fine}$ is the number of fine sites and $B$ the number
of blocks.  The minimization is over unrestricted real fields. In particular,
for any $V \in \R^B$ with $\phi_V$ the minimizer of $\mathcal E_V$,
\begin{equation}
  \mathcal E_V(\phi)
  = \min \mathcal E_V + \frac12 \langle \phi-\phi_V, \operatorname{Hess} \mathcal E_V(0)(\phi-\phi_V)\rangle 
  = \frac12 \langle V, K_{h+1} V\rangle + \frac12 \langle \phi-\phi_V, \operatorname{Hess} \mathcal E_V(0)(\phi-\phi_V)\rangle 
  .
\end{equation}
\end{proposition}

\begin{proof}
Set $A=K_h+aQ^*Q$. For $a>0$, this matrix is positive definite whenever $\ker K_h\cap\ker Q=\{0\}$.
The positive gradient term in $K_h$ makes its kernel consist of constants, while
$Q$ preserves constants, so this condition holds.
Expanding the energy gives
\[
 \mathcal E_V(\phi)
 =\frac12\langle\phi,A\phi\rangle
  -a\langle\phi,Q^*V\rangle+\frac a2\|V\|^2.
\]
Its unique minimizer therefore satisfies
\[
 A\phi_V=aQ^*V,\qquad
 \phi_V=a(K_h+aQ^*Q)^{-1}Q^*V.
\]
Completing the square yields
\[
 \mathcal E_V(\phi)
 =\frac12\langle\phi-\phi_V,A(\phi-\phi_V)\rangle
  +\frac12\langle V,(a\Id-a^2QA^{-1}Q^*)V\rangle.
\]
The first term vanishes at $\phi=\phi_V$, giving the stated formula
for $K_{h+1}$.
\end{proof}


Note that $K_h$ is not finite range when $h \neq \varnothing$.
A sequence of block spin transformations is called \emph{history}
and the corresponding kinetic term $K_h$.


\subsection{Lifting the energy}

\begin{proposition}
The renormalized kinetic term can be written as 
\begin{equation}
 K_h
 =d^* \mathsf A_h^* \mathsf A_h d
 = c_0d^*d+d^*\ell_h^*\ell_hd,
\end{equation}
so that
\begin{equation}
\frac12\langle q,K_hq\rangle
=\frac{c_0}{2}\|dq\|^2+\frac12\|\ell_hdq\|^2,
\end{equation}
where the matrices $\ell_h$ and $\mathsf A_h$ are indexed by edges:
\begin{equation}
  \ell_h : \R^E \to \R^E,
  \qquad
  \mathsf A_h = \begin{pmatrix} \sqrt{c_0} I \\\ \ell_h \end{pmatrix} : \R^E \to \R^{E\times E}.
\end{equation}
\end{proposition}

\begin{proof}
By [O3, Section~4.2], $cD\le K_h\le CD$ and $K_h(p)-D(p)=O(|p|^4)$ in Fourier space, where
$d(p)=\binom{e^{ip_1}-1}{e^{ip_2}-1}$ and $D=d^*d$.
There is $B_h^{\rm raw}=I+O(|p|^2)$ with $K_h=d^*B_h^{\rm raw}d$.
Writing $\mathsf c=(-d_2,d_1)$, we have $\mathsf c d=0$, so
\begin{equation}
 B_h=B_h^{\rm raw}+\lambda_0\mathsf c^*\mathsf c
 \quad\text{satisfies}\quad d^*B_hd=K_h.
\end{equation}
One fixed sufficiently large $\lambda_0$ makes $B_h$ uniformly
positive: near $p=0$ the raw lift is close to $I$; away from $p=0$,
ellipticity controls the longitudinal entry and the added term
controls the transverse entry, including the mixed term by a Schur
complement.  The bounds are uniform in the history.

To use the preprint's particular positive completion [O3, Section~4.3],
make the further modification
\[
 \widetilde B_h^{-1}
 =B_h^{-1}+\lambda_1 B_h^{-1}\mathsf c^*\mathsf c B_h^{-1}.
\]
Since $\widetilde B_h^{-1}B_hd=d$, this still gives
$\widetilde B_hd=B_hd$ and $d^*\widetilde B_hd=K_h$.
Choose $c_0>0$ below the uniform lower bound of $\widetilde B_h$ and set
\[
 \ell_h=(\widetilde B_h-c_0I)^{1/2}.
\]
Then $\mathsf A_h^*\mathsf A_h=c_0I+\ell_h^*\ell_h
=\widetilde B_h$, proving the decomposition and its quadratic-form
identity.  The uniform analyticity and strict positivity imply that
$\ell_h$ has an exponentially decaying kernel.  At empty history the
construction retains $\widetilde B_{\varnothing}=I$.
\end{proof}
There is redundancy between $c_0$ and $\ell_h$.
Initially, for $0<c_0<1$,
\begin{equation}
 K_{\varnothing}=d^*d,
 \qquad \ell_{\varnothing}=\sqrt{1-c_0}\,\Id.
\end{equation}
%
The constant $c_0>0$ is fixed (uniformly over histories) to reserve an explicit positive nearest-neighbour term for local energy estimates.
The operator $\ell_h$ has exponential tail.
Later $\ell_s$ is a spatially truncated version of $\ell_h$ used in the finite range integration.

\begin{proposition} \label{prop:D0}
Define $D_0: \R^{n_{\rm fine}} \to \R^{E \times E \times B}$ by
\begin{equation}
  D_0
  =\begin{pmatrix}
       \sqrt{c_0}\,d\\ \ell_s d\\ -\sqrt a\,Q
  \end{pmatrix}
  =\begin{pmatrix}
       \mathsf A_s d\\ -\sqrt a\,Q
  \end{pmatrix},
 \qquad
 \mathsf A_s=\begin{pmatrix}\sqrt{c_0}\Id\\\ell_s\end{pmatrix}.
\end{equation}
%where
%\begin{equation}
% d,\ell_s d:\R^{n_{\rm fine}}\longrightarrow\R^E,
% \qquad Q:\R^{n_{\rm fine}}\longrightarrow\R^B.
%\end{equation}
Then the Hessian of the variational functional $\mathcal E_V(\phi)$ in \eqref{eq:Gauss-var} can be written as
\begin{equation}
 \|D_0\phi\|^2
 = \langle \phi, D_0^* D_0 \phi \rangle
\end{equation}
and
\begin{equation}
  \mathcal E_V(\phi)- \mathcal E_V(\phi_V)
  = \frac12 \|D_0\phi +c_V - s_V\|^2,
\end{equation}
where $\phi_V$ is the minimizer of $\mathcal E_V$ and, for some $\nu_1,\nu_2,w$,
\begin{equation}
  c_V = \begin{pmatrix}
  0 \\ 0 \\ \sqrt{a}V
  \end{pmatrix},
  \qquad
  s_V
  = \begin{pmatrix}
    \nu_1 \\ \nu_2 \\ w
  \end{pmatrix}
  .
\end{equation}
\end{proposition}

\begin{proof}
The Hessian of the variational functional $\mathcal E_V(\phi)$ in \eqref{eq:Gauss-var} can be written as
\begin{equation}
 \|D_0\phi\|^2
 = \langle \phi, D_0^* D_0 \phi \rangle
 =c_0\sum_e|(d\phi)_e|^2
  +\sum_e|(\ell_s d\phi)_e|^2
  +a\sum_Y|(Q\phi)_Y|^2
\end{equation}
and the full Gaussian energy is
\begin{equation}
  \mathcal E_V(\phi) = \frac12 \|D_0\phi + c_V\|^2 ,
  \qquad
  c_V = \begin{pmatrix}
    0 \\ 0 \\ \sqrt{a} V 
  \end{pmatrix}.
\end{equation}
Let $\phi_V = \operatorname{argmin} \mathcal E_V(\phi)$. Then 
\begin{equation}
  \mathcal E_V(\phi_V + \eta) = \frac12 \|D_0\eta + s_V\|^2  = \frac12 \|D_0\eta\|^2 + \frac12 \|s_V\|^2,
  \qquad
  s_V
  = D_0\phi_V + c_V
  = \begin{pmatrix}
    \nu_1 \\ \nu_2 \\ w
  \end{pmatrix}
\end{equation}
since $D^*s_V=0$. In other words,
\begin{equation}
  \mathcal E_V(\phi)- \mathcal E_V(\phi_V)
  = \frac12 \|D_0\phi - D_0\phi_V\|^2
  = \frac12 \|D_0\phi +c_V - s_V\|^2
\end{equation}
\end{proof}

\section{Nonlinear background field coordinates}
\label{sec:background-field}

This explains Section~5.2.

\subsection{Fluctuation integral}

Throughout this section, $S\subset \Lambda$ are the nonfrozen fine sites (small field region) with
 one spin $q_x \in S^2$ integrated at each site $x\in S$. 
Their number is $n=|S|$.
For each site $x\in S$, there is a background field of reference spins $T_x\in  S^2$
which are held fixed. We consider the integral
\begin{equation}
 \mathcal I=
 \int_{(S^2)^S} H(q)
    \exp\!\left\{-\frac{|R(q)|^2}{2g^2}\right\}
    \prod_{x\in S}d\omega(q_x),
 \label{eq:spin-integral}
\end{equation}
where $|R(q)|^2$ is a reference energy and $d\omega$ the normalized measure on $S^2$.
The function $H(q)$ contains nonlocal contributions not explicitly tracked in $|R(q)|^2$.


\subsection{Reference energy}

The precise form of the reference energy $|R(q)|^2$ is unimportant for the algebraic manipulations
in the subsequent subsections, but it motivated their choice.
$|R(q)|^2$ is a nonlinear version of the Gaussian energy with soft block spin condition,
defined in the last section:
\begin{equation}
  \mathcal E_V(\phi) - \mathcal E_V(\phi_V) = \frac12 \|D_0\phi+c_V-s_V\|^2.
\end{equation}
More precisely, first restrict $D_0: \R^{n} \to \R^{E \times E \times B}$ 
to nonfrozen sites $S \subset \Lambda$ and `selected' components of the range $R\subset E\times E \times B$
(selecting small gradients or so):
\begin{equation}
D = D_0|_{S\times R}.
\end{equation}
%
For the purpose of the discussion, we ignore this restriction; it does not change much.
Thus, up to dropping rows and columns, in Proposition~\ref{prop:D0}, it was shown that
\begin{equation}
D\phi +c_V -s_V = \begin{pmatrix}
  \sqrt{c_0} d\phi\\
  \ell_s d\phi\\
  \sqrt{a}V -\sqrt{a}Q\phi
\end{pmatrix}
-s_V,
  \qquad
  s_V = \begin{pmatrix}
    \nu_1 \\ \nu_2 \\ w
  \end{pmatrix}
.
\end{equation}
%
For $q: \R^n  \to S^2$, define $R(q)$ as a nonlinear version of this:
\begin{equation}
  R(q) = \begin{pmatrix}
  \sqrt{c_0} dq\\
  \ell_s dq\\
  \sqrt{a}(V-\widehat m(q))
  \end{pmatrix}
  - \begin{pmatrix}
    \nu_1 \\ \nu_2 \\ w
  \end{pmatrix}
\end{equation}
where we recall that $\widehat m_Y(q)$ is the nonlinear block average \eqref{eq:hatm}.

Concretely, the quantity $\tfrac12|R(q)|^2$ arises as follows.
The renormalized density representation \eqref{eq:density} describes the full density before or after that
step.  The connection is obtained by extracting the coarse kinetic
energy from the fine kinetic energy plus the block-spin observation
term.  Write \eqref{eq:density} as
\begin{equation}
 \rho_j(q)=e^{\mathfrak v_j|\Lambda_j|}
          e^{-b\mathcal E_{h,t}(q)}G_j(q).
\end{equation}
The block transformation integrates this against the block spin
kernel whose exponential part is
\begin{equation}
 e^{-\frac{ba}{2}\|V-\widehat m(q)\|^2}.
\end{equation}
Then the leading part of the energy decomposition has the following form:
\begin{equation}
 \mathcal E_{h,t}(q)
 +\frac a2\|V-\widehat m(q)\|^2
 =\mathcal E_{h',t'}(V)
  +\frac12|R(q;V)|^2+\mathcal L(q,V).
\end{equation}
Here $h',t'$ are the next-scale history and large field scale, $R(q;V)$
denotes the selected residual rows defined above and $\mathcal L(q,V)$ collects the remainder.

\subsection{Intrinsic tangent coordinates}

Consider a single site and reference spin $T \in S^2$.
For $u\in T^\perp$,
\begin{equation}
 \Exp_T u=\cos|u|\,T+\frac{\sin|u|}{|u|}u,
 \qquad \Exp_T0=T.
\end{equation}
The restriction $|u|<\pi$ parametrizes $S^2\setminus\{-T\}$ and the omitted point has area zero.
Note
\begin{equation}
 d\omega(\Exp_Tu)=J(u)\,d^2u,
 \qquad J(u)=\frac1{4\pi}\frac{\sin|u|}{|u|},
 \qquad J(0)=\frac1{4\pi}.
 \label{eq:area}
\end{equation}
In tangent coordinates, 
$q(u)=(\Exp_{T_x}u_x)_{x\in S}$ and
\begin{equation}
 \mathcal U=\prod_{x\in S}
       \{u_x\in T_x^\perp:|u_x|<\pi\}.
\end{equation}
The spherical area formula gives
\begin{equation}
 \mathcal I=
 \int_{\mathcal U} H(q(u))
 \exp\!\left\{-\frac{|R(q(u))|^2}{2g^2}\right\}
 \prod_{x\in S}\bigl[J(u_x)\,d^2u_x\bigr].
\end{equation}
Here $d^2u_x$ is Euclidean area measure on $T_x^\perp$.
This integral has $2n$ real coordinates.

\subsection{Auxiliary normal coordinate}

Introduce one real variable $v_x\in\R$ at each nonfrozen site.
Since $A=D^*D$ is positive definite, the elementary
Gaussian integral gives
\begin{equation}
 1=\frac{(\det A)^{1/2}}{(2\pi g^2)^{n/2}}
       \int_{\R^n}\exp\!\left\{-\frac{v^*Av}{2g^2}\right\}d^nv.
 \label{eq:scalar-one}
\end{equation}
Equivalently, the inserted probability law is a centered scalar
Gaussian with covariance $g^2A^{-1}$.  Its energy is
$v^*Av=|Dv|^2$.  No cutoff is imposed on $v$, and
the inserted variables are initially independent of the spin variables.
Thus
\begin{equation}
 \begin{aligned}
 \mathcal I={}&\frac{(\det A)^{1/2}}{(2\pi g^2)^{n/2}}
 \int_{\R^n}\int_{\mathcal U} H(q(u))\\
 &\quad\times
 \exp\!\left\{-\frac{|R(q(u))|^2+|Dv|^2}{2g^2}\right\}
 \prod_{x\in S}\bigl[J(u_x)\,d^2u_x\, dv_x\bigr].
 \label{eq:uv-integral}
 \end{aligned}
\end{equation}
There are now $2n+n=3n$ real integration coordinates.
The purpose of the extra coordinate is
to complete each tangent plane to the ambient space via
\begin{equation}
\xi_x=u_x+T_xv_x
.
\end{equation}
Thus the scalar $v_x$ is an auxiliary normal coordinate, not a new physical spin component.

\subsection{Combining the coordinates into an ambient field}

Conversely, for $\xi=(\xi_x)_{x\in S}\in(\R^3)^S$, define the tangential and normal directions
\begin{equation}
 u_x(\xi)=\xi_x-(T_x\cdot\xi_x)T_x,
 \qquad
 v_x(\xi)=T_x\cdot\xi_x
\end{equation}
so that
\begin{equation}
 q_x(\xi)=\Exp_{T_x}u_x(\xi).
\end{equation}
The change of variables $(u_x,v_x) \mapsto \xi_x=u_x+T_xv_x$ has Jacobian one.
Thus
\begin{equation}
 \mathcal I=
 \frac{(\det A)^{1/2}}{(2\pi g^2)^{n/2}}
 \int_{\R^{3n}} H(q(\xi))\,\mathcal J(\xi)
 \exp\!\left\{-\frac{|R(q(\xi))|^2+|Dv(\xi)|^2}{2g^2}\right\}
 \,d^{3n}\xi ,
\end{equation}
where the chart factor is
\begin{equation}
 \mathcal J(\xi)=\prod_{x\in S} J(u_x(\xi)) 1_{|u_x(\xi)| < \pi}.
 %j(u)=
 %\begin{cases}
 %  \sin|u|/|u|,& |u|<\pi,\\
 %  0,& |u|\ge\pi,
 %\end{cases}
 %\qquad j(0)=1.
\end{equation}
%The factor $J(0)^n\mathcal J(\xi) = (4\pi)^{-n} \mathcal J(\xi)$ supplies the spherical
%Jacobian and the principal-chart restriction, while
%$(\det A)^{1/2}/(2\pi g^2)^{n/2}$ normalizes the auxiliary normal Gaussian.

\subsection{Gaussian coordinates}

Recall that $D: (\R^{3})^n \to (\R^3)^m$ acts on the nonfrozen coordinates.
Let $F=(F_i)_{i=1}^m\in(\R^3)^m$ be a fixed vector in the range so that
\begin{equation}
 (D\xi+F)_i=\sum_{x\in S}D_{ix}\xi_x+F_i.
\end{equation}
%
Define 
\begin{equation}
 \Delta(\xi)=|R(q(\xi))|^2+|Dv(\xi)|^2-|D\xi+F|^2.
 \label{eq:delta}
\end{equation}
Note that $R(q(\xi))$ is defined in terms of the physical sphere valued coordinates
whereas $D\xi+F$ is the affine ambient expression chosen for the Gaussian computation.
%
Clearly,
\begin{equation}
 e^{-(|R(q(\xi))|^2+|Dv|^2)/(2g^2)}
 =e^{-|D\xi+F|^2/(2g^2)}e^{-\Delta(\xi)/(2g^2)}.
\end{equation}
The first term on the right-hand side is a Gaussian density with
covariance $g^2A^{-1}$ where $A=D^* D$.


\subsection{Gaussian expectation identity}

It will be useful to choose a Gaussian reference measure with more general covariance $g^2C$ than $g^2 A^{-1}$.
Choose any symmetric positive definite $n\times n$ matrix $C$
and any $\mu\in(\R^3)^S$.
Let $Z\in(\R^3)^S$ have the normalized Gaussian density with
\begin{equation}
\E[Z_x^\alpha Z_y^\beta]=C_{xy}\delta_{\alpha\beta}.
\end{equation}
%
The vector $\mu\in(\R^3)^S$ is an arbitrary fixed shift in
the Gaussian coordinates
\begin{equation}
 \xi=\mu+gZ,\qquad Z\sim N(0,C\otimes I_3).
\end{equation}
The covariance $C$ has not been determined yet, but it will have finite range.

\begin{remark}
The unique minimizer of the Gaussian action
\begin{equation}
\mathcal E_F(\xi)=
\frac12|D\xi+F|^2
\end{equation}
is 
\begin{equation}
\xi = \mu_*=-A^{-1}D^*F.
\end{equation}
\end{remark}

\begin{proof}
\begin{equation}
 \nabla_\xi\frac12|D\xi+F|^2=A\xi+D^*F,
\end{equation}
so positive definiteness of $A$ makes $\mu_*$ the unique minimizer. 
\end{proof}

This above choice $\mu=\mu_*$ gives $h=0$ in the identity below, eliminating the term linear in $Z$ in the exponent in the following proposition.
The general identity below permits other choices. The preprint uses a finite range approximation
\begin{equation}
 \mu_s=-\Pi_0^*D^*F,
\end{equation}
related to the choice $C=C_s$ where
\begin{equation}
  C_s = \frac12 (\Pi_0+\Pi_0^*),
\end{equation}
where the finite range condition is enforced by
\begin{equation}
  (\Pi_{a_0})_{yx} = 1_{y\in\Omega_x} (A_{\Omega_x} +a_0)^{-1}_{yx}.
\end{equation}
For this choice, $h=A\mu_s+D^*F$ need not vanish;
its contribution is retained explicitly.

\begin{proposition} \label{prop:background-expect}
Given arbirary $F$, for any $C$ and $\mu$,
\begin{align}
 \mathcal I={}&
 \kappa_g^n(\det A)^{1/2}(\det C)^{3/2}
 \exp\!\left\{-\frac{|r|^2}{2g^2}\right\}
 \nonumber\\
 &\E_{\gamma_C}\left[
 H(q(\mu+gZ))\,\mathcal J(\mu+gZ)
 \exp\!\left\{
 -\frac{\Delta(\mu+gZ)}{2g^2}
 +\frac12 Z^*(C^{-1}-A)Z
 -g^{-1}(h,Z)
 \right\}\right],
 \label{eq:main}
\end{align}
where %$h^*Z=\sum_xh_x\cdot Z_x$ and
\begin{equation}
 r=D\mu+F,\qquad h=A\mu+D^*F,
 \qquad \kappa_g=(2\pi g^2)=\frac{g^2}{2}.
\end{equation}
\end{proposition}

\begin{proof}
By the above,
\begin{align}
 \mathcal I={}&
 \frac{(\det A)^{1/2}}{(2\pi g^2)^{n/2}}
 \int_{\R^{3n}}H(q(\xi))\,\mathcal J(\xi)
 \exp\!\left\{-\frac{|D\xi+F|^2+\Delta(\xi)}{2g^2}\right\}d^{3n}\xi.
 \label{eq:lebesgue}
\end{align}
Now set $\xi=\mu+gZ$, whose Jacobian is $g^{3n}$.
Ordinary multiplication of squares gives
\begin{equation}
 |D(\mu+gZ)+F|^2
 =|r|^2+2g\,h^*Z+g^2Z^*AZ.
\end{equation}
The constant is
\begin{equation}
 \frac{(\det A)^{1/2}}{(2\pi g^2)^{n/2}}
 g^{3n}(2\pi)^{3n/2}(\det C)^{3/2}
 =\kappa_g^n(\det A)^{1/2}(\det C)^{3/2}.
\end{equation}
\end{proof}

\section{Parametrization of the renormalized action}

The renormalized action or more precisely its exponential (called `retained density') is parametrized in [O3, (3.29)].
The density is relative to product normalized spherical area measure $\prod_xd\omega(q_x)$.
Its representation is
\begin{equation}
\begin{aligned}
 \rho_j(q)={}&
 e^{\mathfrak v|\Lambda_j|-b\mathcal E_{h,t}(q)}\,
 \exp\!\left\{
 -\sum_{\alpha}
       1_{X_\alpha}(q)\,\mathcal P_\alpha(q)
 -\sum_X1_X(q)f_X(q)\right\}\,\Xi(q).
 \end{aligned}
 \label{eq:density}
\end{equation}
The different terms describe:
\begin{itemize}
\item $\mathfrak v$ is the partition function renormalization (a constant);
\item $b \mathcal E_{h,t}$ is a regularized version of the kinetic energy. If all gradients of $q$
are small then
\begin{equation}
\mathcal E_{h,t}(q) = \frac12 \langle q, K_h q\rangle
\end{equation}
and $t$ parametrizes a cutoff scale for the size of gradients.

This term is marginal and has size $g^0$. Indeed, its prefactor $b\approx 1/g^2$ and fluctuations are of size $dq^2 \approx g^2$.

\item $\mathcal P_\alpha$ are irrelevant second- to sixth-order monomials, of formal orders $g^2$ and $g^4$.
\item $f$ are the remaining irrelevant terms of size $\approx g^{4.1}$ (smaller than second order in $g^2$). 
\item $\Xi$ are nonperturbative large field terms (smaller than $e^{-(log g)^\epsilon}$).
\end{itemize}
%
Given a field scale parameter $t>0$:
\begin{itemize}
\item 
The large field region (set of bad edges) is defined by
\begin{equation}
 D(q)=\{e: |d_e q|>t\}.
 \label{eq:bad-edges}
\end{equation}
\item
The indicator function that $X$ is in the small field region (only has good edges) is
\begin{equation}
 1_X(q)=\mathbf1\{|d_eq| \le  t\text{ for every }e\in X\}.
 \label{eq:mask}
\end{equation}
\end{itemize}
 The parameter is chosen as $g_j \ll t \ll g_t^{1-epsilon}$. Typical fluctuations should be of order $g_j$.

\subsection{Renormalized kinetic energy $\mathcal E$}

The renormalized kinetic term is defined by
\begin{equation}
\exp\big\{-b\mathcal E_{h,t}(q) \big\}.
\end{equation}
Given fixed $0<d_0<c_0/16$ and $R_* \gg 1$, for any large field scale $t>0$, define
\begin{align}
 S_t(r)&=\begin{cases}
 c_0r^2/2,&r\le t,\\
 d_0tr,&r>t,
 \end{cases}\\
 m_{e,t}(q)&=\mathbf1\!\left\{
 |dq_f|\le t\,e^{\operatorname{dist}(e,f)/R_*}
                       \text{ for every edge }f\right\},
\end{align}
and the regularized kinetic energy
\begin{equation}
 \mathcal E_{h,t}(q)
 =
 \sum_e S_t(|d_e(q)|)+
 \frac12\sum_em_e(q)|(\ell_hdq)_e|^2.
 \label{eq:energy}
\end{equation}
The constant $c_0$ and the edge operator $\ell_h$ are those from the Gaussian decomposition in Section~\ref{sec:block-gauss}:
\begin{equation}
 K_h=c_0d^*d+d^*\ell_h^*\ell_hd.
\end{equation}
If every edge is good, then $m_e=1$ for every $e$ and
\begin{equation}
 \mathcal E_{h,t}(q)=\frac12\langle q,K_hq\rangle.
\end{equation}
Thus this is a nonlinear version of the Gaussian reference measure that we typically use in RG.

\subsection{High order irrelevant terms $f$}

The contribution is
\begin{equation}
  \exp\left\{-\sum_X 1_X(q) f_X(q) \right\}.
\end{equation}
The actitivies $f_X$ are defined for connected $X$ with support and have norms
\begin{equation}
  \|f\| = \sup_o \sum_{X \ni o} [f_X].
\end{equation}
The brackets denote the scaled local $C^k$ norm
\begin{equation}
 [f_X]_{k,j}
 =\sum_{r=0}^{k}
   \sup_{q\in\mathcal U_{X}}
   \sup_{v_1,\ldots,v_r}
   |D_{v_1,\ldots,v_r}f_X(q)|,
 \qquad 0\le k\le8.
 \label{eq:error-seminorm}
\end{equation}
where $\mathcal U_X$ is a set of spins containing the support of $1_X$.
Writing $v\times$ for the skew-symmetric linear map
$w\mapsto v\times w$ on $\R^3$, the rotation derivatives are
\begin{equation}
 D_{v_1,\ldots,v_r}f_X(q)
 =\left.
   \partial_{\theta_1}\cdots\partial_{\theta_r}
   f_X\!\left(\left(
     \exp\!\left[\left(\sum_{a=1}^{r}\theta_av_{a,x}\right)
                      \times\right]q_x
   \right)_x\right)
   \right|_{\theta=0}.
 \label{eq:error-rotation-jets}
\end{equation}
The supremum runs over vector fields satisfying
\begin{equation}
 |v_{a,x}|\le t(1+\operatorname{dist}(x,o))^8.
 \label{eq:error-directions}
\end{equation}
The rotations keep the spins on $S^2$.
Each derivative is scaled by $t$, so schematically this norm resembles
$\sum_{r=0}^k t^r\|\nabla^r f_X\|_\infty$, with spatial weights
and derivatives adapted to the sphere.  The exact definition is
[O3, (3.20)--(3.22)].

\subsection{Nonperturbative terms $\Xi$}

The exact sum is
\begin{equation}
 \Xi(q)=\sum_{\substack{\Gamma\text{ a finite family of labels}\\
       P_\ell\cap P_{\ell'}=\varnothing\ (\ell\ne\ell')\\
       \bigsqcup_{\ell\in\Gamma}J_\ell=D(q)}}
                    \prod_{\ell\in\Gamma}k_\ell(q).
 \label{eq:covering}
\end{equation}
The polymer activities satisfy:
\begin{itemize}
  \item 
Each $k_\ell(q)$ is supported in the large field region:
\begin{equation}
 k_\ell(q)=0\quad\text{unless}\quad J_\ell\subset D(q).
\end{equation}
In particular, if $D(q)=\varnothing$ (there are no large fields), then $\Xi(q)=1$.
\item
The size of the $k_\ell$ is measured in $L^\infty$ rather than $C^8$:
\begin{equation}
 \|k\|
 = \sup_x\sum_{\ell:x\in P_\ell}
 e^{64s_\ell}\|k_\ell\|_\infty
 \label{eq:cover-bound}
\end{equation}
and eventually $\|k\| \le w=e^{-p^{1/4}}$.
\end{itemize} 

\subsection{Low order irrelevant terms $\mathcal P$}

These are essentially irrelevant sixth order polynomials in the fields,
written in tangent coordinates. They are defined in [O3, (3.13)--(3.19)]. 

The relative tangent projection of $q_x$ at an anchor $o$ is
\begin{equation}
 y_o(x)=q_x-(q_o\cdot q_x)q_o\in q_o^\perp.
 \label{eq:canonical-tangent}
\end{equation}
This is a projection, not the logarithmic coordinate $\Log_{q_o}q_x$.
A monomial label $\alpha$ consists of a degree $n_\alpha$, a coefficient
$a_\alpha$, displacements $r_1^\alpha,\ldots,r_n^\alpha\in\mathbb Z^2$, and an

\begin{equation}
 a_\alpha T_\alpha\bigl(y_o(o+r_1^\alpha),\ldots,y_o(o+r_n^\alpha)\bigr).
\end{equation}
The set $X_\alpha$ is a connected graph containing the anchor $o$
the $r_1,\dots,r_n$ with the joining paths and enlargements prescribed by its construction.

The tensors $T_\alpha$ are chosen from a fixed basis of the invariant tensor space.
They are evaluated in $q_o^\perp$ through any orthonormal identification
with $\R^2$; invariance makes the answer independent of that choice.
